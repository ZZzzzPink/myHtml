# 在线音乐播放网站开发文档

## 1. 项目概述

### 1.1 项目简介
基于现有Vue3项目框架，集成音乐播放API，开发一个兼容Web和H5的在线音乐播放网站。项目采用渐进式开发策略，先实现核心功能，后续可扩展高级特性。

### 1.2 技术栈
- **前端框架**: Vue 3 + Composition API
- **构建工具**: Vite
- **UI组件库**: 
  - Element Plus (主要桌面端)
  - Vant (移动端H5)
  - Ant Design Vue (辅助组件)
- **状态管理**: Pinia + pinia-plugin-persist
- **路由**: Vue Router 4 (Hash模式)
- **HTTP客户端**: Axios
- **样式**: UnoCSS + SCSS
- **音频播放**: HTML5 Audio API
- **开发工具**: VConsole (移动端调试)

### 1.3 支持的音乐平台
- 网易云音乐 (`netease`)
- 酷我音乐 (`kuwo`) 
- QQ音乐 (`qq`)

### 4.3 错误处理和重试机制
```javascript
// src/api/music/utils.js
export class ApiError extends Error {
  constructor(message, code, data) {
    super(message)
    this.name = 'ApiError'
    this.code = code
    this.data = data
  }
}

export const withRetry = async (apiCall, maxRetries = 3) => {
  let lastError
  
  for (let i = 0; i < maxRetries; i++) {
    try {
      return await apiCall()
    } catch (error) {
      lastError = error
      if (i < maxRetries - 1) {
        await new Promise(resolve => setTimeout(resolve, 1000 * (i + 1)))
      }
    }
  }
  
  throw lastError
}
```

```javascript
// src/composables/music/useTouchGesture.js
import { ref, onMounted, onUnmounted } from 'vue'

export function useTouchGesture(elementRef) {
  const startX = ref(0)
  const startY = ref(0)
  const isDragging = ref(false)

  const handleTouchStart = (e) => {
    startX.value = e.touches[0].clientX
    startY.value = e.touches[0].clientY
    isDragging.value = true
  }

  const handleTouchMove = (e) => {
    if (!isDragging.value) return
    
    const deltaX = e.touches[0].clientX - startX.value
    const deltaY = e.touches[0].clientY - startY.value
    
    // 处理滑动手势
    if (Math.abs(deltaX) > Math.abs(deltaY)) {
      // 水平滑动 - 切换歌曲
      if (Math.abs(deltaX) > 50) {
        handleSwipe(deltaX > 0 ? 'next' : 'prev')
        isDragging.value = false
      }
    }
  }

  const handleTouchEnd = () => {
    isDragging.value = false
  }

  onMounted(() => {
    const element = elementRef.value
    element?.addEventListener('touchstart', handleTouchStart)
    element?.addEventListener('touchmove', handleTouchMove)
    element?.addEventListener('touchend', handleTouchEnd)
  })

  onUnmounted(() => {
    const element = elementRef.value
    element?.removeEventListener('touchstart', handleTouchStart)
    element?.removeEventListener('touchmove', handleTouchMove)
    element?.removeEventListener('touchend', handleTouchEnd)
  })

  return { startX, startY, isDragging }
}
```

```javascript
// src/composables/music/usePerformance.js
import { ref, onMounted, onUnmounted } from 'vue'

export function usePerformance() {
  const isLowEndDevice = ref(false)
  const reducedMotion = ref(false)

  const detectDevice = () => {
    // 检测设备性能
    const connection = navigator.connection || navigator.mozConnection || navigator.webkitConnection
    const hardwareConcurrency = navigator.hardwareConcurrency || 4
    const deviceMemory = navigator.deviceMemory || 4

    // 判断是否为低端设备
    isLowEndDevice.value = (
      hardwareConcurrency <= 4 ||
      deviceMemory <= 4 ||
      (connection && connection.effectiveType && ['slow-2g', '2g', '3g'].includes(connection.effectiveType))
    )

    // 检测减少动画偏好
    reducedMotion.value = window.matchMedia('(prefers-reduced-motion: reduce)').matches
  }

  onMounted(() => {
    detectDevice()
    
    // 监听网络变化
    const connection = navigator.connection
    if (connection) {
      connection.addEventListener('change', detectDevice)
    }
  })

  return { isLowEndDevice, reducedMotion }
}
```

```javascript
import { ref } from 'vue'

export function useMobileAudio() {
  const isMobile = /iPhone|iPad|iPod|Android/i.test(navigator.userAgent)
  const requiresUserInteraction = ref(isMobile)
  const hasInteracted = ref(false)

  const enableAudio = async () => {
    if (requiresUserInteraction.value && !hasInteracted.value) {
      // 创建一个静音的音频上下文来启用音频
      const audioContext = new (window.AudioContext || window.webkitAudioContext)()
      await audioContext.resume()
      hasInteracted.value = true
      requiresUserInteraction.value = false
    }
  }

  const handleFirstInteraction = () => {
    enableAudio()
    document.removeEventListener('touchstart', handleFirstInteraction)
    document.removeEventListener('click', handleFirstInteraction)
  }

  // 监听首次用户交互
  if (requiresUserInteraction.value) {
    document.addEventListener('touchstart', handleFirstInteraction, { once: true })
    document.addEventListener('click', handleFirstInteraction, { once: true })
  }

  return { requiresUserInteraction, hasInteracted, enableAudio }
}
```

```javascript
// src/stores/music/player.js
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

export const usePlayerStore = defineStore('player', () => {
  // 状态
  const currentSong = ref(null)
  const isPlaying = ref(false)
  const currentTime = ref(0)
  const duration = ref(0)
  const volume = ref(1)
  const isMuted = ref(false)
  const playMode = ref('sequence') // sequence, random, single
  const audioElement = ref(null)

  // 计算属性
  const progress = computed(() => {
    return duration.value > 0 ? currentTime.value / duration.value : 0
  })

  const isPaused = computed(() => !isPlaying.value)

  // 方法
  const setCurrentSong = (song) => {
    currentSong.value = song
    currentTime.value = 0
  }

  const play = () => {
    if (audioElement.value) {
      audioElement.value.play()
      isPlaying.value = true
    }
  }

  const pause = () => {
    if (audioElement.value) {
      audioElement.value.pause()
      isPlaying.value = false
    }
  }

  const togglePlay = () => {
    isPlaying.value ? pause() : play()
  }

  const setCurrentTime = (time) => {
    if (audioElement.value) {
      audioElement.value.currentTime = time
      currentTime.value = time
    }
  }

  const setVolume = (vol) => {
    if (audioElement.value) {
      audioElement.value.volume = vol
      volume.value = vol
      isMuted.value = vol === 0
    }
  }

  const toggleMute = () => {
    const newMuted = !isMuted.value
    if (audioElement.value) {
      audioElement.value.muted = newMuted
      isMuted.value = newMuted
    }
  }

  const setPlayMode = (mode) => {
    playMode.value = mode
  }

  return {
    // 状态
    currentSong,
    isPlaying,
    currentTime,
    duration,
    volume,
    isMuted,
    playMode,
    audioElement,
    // 计算属性
    progress,
    isPaused,
    // 方法
    setCurrentSong,
    play,
    pause,
    togglePlay,
    setCurrentTime,
    setVolume,
    toggleMute,
    setPlayMode
  }
}, {
  persist: {
    key: 'music-player',
    storage: localStorage,
    paths: ['volume', 'isMuted', 'playMode']
  }
})
```

```javascript
// src/stores/music/playlist.js
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

export const usePlaylistStore = defineStore('playlist', () => {
  // 状态
  const playList = ref([])
  const currentIndex = ref(-1)
  const favoriteList = ref([])
  const searchHistory = ref([])

  // 计算属性
  const currentSong = computed(() => {
    return currentIndex.value >= 0 ? playList.value[currentIndex.value] : null
  })

  const hasNext = computed(() => {
    return currentIndex.value < playList.value.length - 1
  })

  const hasPrevious = computed(() => {
    return currentIndex.value > 0
  })

  // 方法
  const addToPlaylist = (songs) => {
    playList.value.push(...songs)
  }

  const removeFromPlaylist = (index) => {
    if (index >= 0 && index < playList.value.length) {
      playList.value.splice(index, 1)
      if (index < currentIndex.value) {
        currentIndex.value--
      }
    }
  }

  const clearPlaylist = () => {
    playList.value = []
    currentIndex.value = -1
  }

  const playAtIndex = (index) => {
    if (index >= 0 && index < playList.value.length) {
      currentIndex.value = index
    }
  }

  const playNext = () => {
    if (hasNext.value) {
      currentIndex.value++
    }
  }

  const playPrevious = () => {
    if (hasPrevious.value) {
      currentIndex.value--
    }
  }

  const addToFavorites = (song) => {
    if (!favoriteList.value.find(s => s.id === song.id)) {
      favoriteList.value.push(song)
    }
  }

  const removeFromFavorites = (songId) => {
    const index = favoriteList.value.findIndex(s => s.id === songId)
    if (index >= 0) {
      favoriteList.value.splice(index, 1)
```

```javascript
export function useAudioPreloader() {
  const preloadQueue = ref([])
  const isPreloading = ref(false)

  const preloadNextSongs = async (currentSong, playList, count = 2) => {
    const currentIndex = playList.findIndex(song => song.id === currentSong.id)
    const nextSongs = playList.slice(currentIndex + 1, currentIndex + 1 + count)
    
    for (const song of nextSongs) {
      try {
        const audio = new Audio()
        audio.preload = 'auto'
        audio.src = song.url
        await new Promise((resolve, reject) => {
          audio.addEventListener('canplaythrough', resolve)
          audio.addEventListener('error', reject)
        })
      } catch (error) {

      }
    }
  }

  return { preloadNextSongs, isPreloading }
```

```javascript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

export const usePlayerStore = defineStore('player', () => {

  const currentSong = ref(null)
  const isPlaying = ref(false)
  const currentTime = ref(0)
  const duration = ref(0)

    currentTime.value = 0

      isPlaying.value = true
    }
  }

    if (audioElement.value) {
      audioElement.value.pause()
      isPlaying.value = false
    }
  }

  const togglePlay = () => {
    isPlaying.value ? pause() : play()
  }

  return {
    currentSong, isPlaying, currentTime, duration, volume, isMuted, playMode, audioElement,
    progress, setCurrentSong, play, pause, togglePlay
```

```vue
    error.value = ''
    if (!searchKeyword.value.trim()) {
      error.value = '请输入搜索关键词'
      return
    }
    
    if (searchKeyword.value.trim().length < 2) {
      error.value = '搜索关键词至少需要2个字符'
      return
    }
    
    // 添加到搜索历史
    playlistStore.addToSearchHistory(searchKeyword.value)
    
    // 触发搜索事件
    emit('search', searchKeyword.value.trim())
    showHistory.value = false
    
  } catch (err) {
    error.value = err.message || '搜索失败'
  }
}

const searchKeyword = (keyword) => {
  searchKeyword.value = keyword
  handleSearch()
}
```
